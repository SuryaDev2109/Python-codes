import pandas as pd
import numpy as np

print("\n========== DATA PROFILING REPORT ==========")

df = pd.read_csv("file path\file name.csv")
df.head()

print("Original Shape:", df.shape)

df.columns = (
    df.columns
    .str.strip()
    .str.lower()
    .str.replace(" ", "_")
)

print("\nColumns:")
print(df.columns.tolist())

df = df.drop_duplicates()

df = df.drop_duplicates(
    subset=["product_id"],
    keep="first"
)

text_columns = [
    "product_id",
    "product_name",
    "category",
    "machine_type",
    "brand",
    "model_compatibility",
    "dimensions_cm",
    "material_type",
    "criticality_level",
    "usage_frequency",
    "uom"
]

for col in text_columns:
    df[col] = (
        df[col]
        .astype("string")
        .str.strip()
    )


for col in [
    "product_name",
    "category",
    "machine_type",
    "brand",
    "model_compatibility",
    "material_type",
    "criticality_level",
    "usage_frequency",
    "uom"
]:
    df[col] = df[col].str.title()

df["product_id"] = df["product_id"].str.upper()

numeric_columns = [
    "unit_cost",
    "unit_price",
    "margin_percentage",
    "gst_rate",
    "weight_kg",
    "warranty_months",
    "reorder_level",
    "safety_stock",
    "max_stock_level",
    "lead_time_days",
    "last_purchase_price"
]

for col in numeric_columns:
    df[col] = pd.to_numeric(
        df[col],
        errors="coerce"
    )

df["last_purchase_date"] = pd.to_datetime(
    df["last_purchase_date"],
    errors="coerce"
)

for col in text_columns:
    df[col] = df[col].fillna("Unknown")

for col in numeric_columns:
    df[col] = df[col].fillna(
        df[col].median()
    )

positive_columns = [
    "unit_cost",
    "unit_price",
    "weight_kg",
    "warranty_months",
    "reorder_level",
    "safety_stock",
    "max_stock_level",
    "lead_time_days",
    "last_purchase_price"
]

for col in positive_columns:

    df.loc[
        df[col] < 0,
        col
    ] = np.nan

    df[col] = df[col].fillna(
        df[col].median()
    )

invalid_gst = (
    (df["gst_rate"] < 0) |
    (df["gst_rate"] > 100)
)

print("\nInvalid GST Rates:")
print(invalid_gst.sum())

df.loc[
    invalid_gst,
    "gst_rate"
] = np.nan

df["gst_rate"] = df[
    "gst_rate"
].fillna(
    df["gst_rate"].median()
)

invalid_margin = (
    (df["margin_percentage"] < 0) |
    (df["margin_percentage"] > 100)
)

print("\nInvalid Margin Percentages:")
print(invalid_margin.sum())

df.loc[
    invalid_margin,
    "margin_percentage"
] = np.nan

df["margin_percentage"] = df[
    "margin_percentage"
].fillna(
    df["margin_percentage"].median()
)

invalid_price = (
    df["unit_price"] < df["unit_cost"]
)

print("\nProducts where Unit Price < Unit Cost:")
print(invalid_price.sum())

df["margin_percentage"] = np.where(
    df["unit_price"] > 0,
    (
        (df["unit_price"] - df["unit_cost"])
        / df["unit_price"]
    ) * 100,
0
)

invalid_safety = (
    df["safety_stock"] >
    df["reorder_level"]
)

print("\nInvalid Safety Stock:")
print(invalid_safety.sum())

df.loc[
    invalid_safety,
    "safety_stock"
] = df.loc[
    invalid_safety,
    "reorder_level"
]

invalid_reorder = (
    df["reorder_level"] >
    df["max_stock_level"]
)

print("\nInvalid Reorder Level:")
print(invalid_reorder.sum())

df.loc[
    invalid_reorder,
    "reorder_level"
] = df.loc[
    invalid_reorder,
    "max_stock_level"
]

df.loc[
    df["lead_time_days"] < 0,
    "lead_time_days"
] = 0

df.loc[
    df["warranty_months"] < 0,
    "warranty_months"
] = 0

df["unit_cost"] = df["unit_cost"].round(2)
df["unit_price"] = df["unit_price"].round(2)
df["margin_percentage"] = (
    df["margin_percentage"].round(2)
)
df["weight_kg"] = df["weight_kg"].round(2)
df["last_purchase_price"] = (
    df["last_purchase_price"].round(2)
)

df = df[
    df["product_id"] != "UNKNOWN"
]


print("\n========== DATA QUALITY REPORT ==========")

print("\nFinal Shape:")
print(df.shape)

print("\nMissing Values:")
print(df.isnull().sum())

print("\nDuplicate Rows:")
print(df.duplicated().sum())

print("\nDuplicate Product IDs:")
print(
    df["product_id"].duplicated().sum()
)

print("\nData Types:")
print(df.dtypes)

print("\nCriticality Levels:")
print(
    df["criticality_level"].value_counts()
)

print("\nUsage Frequency:")
print(
    df["usage_frequency"].value_counts()
)

print("\nUOM:")
print(
    df["uom"].value_counts()
)

df.to_csv(
    "file name_cleaned.csv",
    index=False
)
